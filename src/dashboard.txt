function escHtml(s) {
  return String(s == null ? '' : s)
    .split('&').join('&amp;')
    .split('<').join('&lt;')
    .split('>').join('&gt;')
    .split('"').join('&quot;')
    .split("'").join('&#39;');
}

async function api(path, method, body) {
  var res = await fetch(PANEL + path, {
    method: method || 'GET',
    headers: body ? { 'content-type': 'application/json' } : {},
    body: body || undefined,
    credentials: 'same-origin',
  });
  var data = await res.json().catch(function () { return {}; });
  if (!res.ok) throw new Error(data.error || ('HTTP ' + res.status));
  return data;
}

function fmtBytes(n) {
  n = Number(n) || 0;
  var u = ['B', 'KB', 'MB', 'GB', 'TB'];
  var i = 0;
  while (n >= 1024 && i < u.length - 1) { n /= 1024; i++; }
  return n.toFixed(i ? 1 : 0) + ' ' + u[i];
}

function fmtDate(ts) {
  if (!ts) return '∞';
  try { return new Date(ts * 1000).toLocaleDateString('fa-IR'); } catch (e) { return String(ts); }
}

function toast(msg, ok) {
  var t = document.createElement('div');
  t.className = 'toast' + (ok === false ? ' bad' : '');
  t.textContent = msg;
  document.body.appendChild(t);
  setTimeout(function () { t.classList.add('in'); }, 10);
  setTimeout(function () { t.classList.remove('in'); setTimeout(function () { t.remove(); }, 300); }, 2200);
}

function statePill(u) {
  if (u.disabled) return '<span class="pill off">غیرفعال</span>';
  if (u.expire && Math.floor(Date.now() / 1000) > u.expire) return '<span class="pill warn">منقضی</span>';
  if (u.limit_bytes > 0 && (u.used_bytes || 0) >= u.limit_bytes) return '<span class="pill warn">اتمام حجم</span>';
  return '<span class="pill ok">فعال</span>';
}

function copyText(t) {
  if (navigator.clipboard && navigator.clipboard.writeText) {
    navigator.clipboard.writeText(t).then(function () { alert('کپی شد ✓'); }, function () { fallbackCopy(t); });
  } else fallbackCopy(t);
}

function fallbackCopy(t) {
  var ta = document.createElement('textarea');
  ta.value = t;
  document.body.appendChild(ta);
  ta.select();
  document.execCommand('copy');
  ta.remove();
  toast('کپی شد ✓');
}

async function loadCatalog(force) {
  catLoading = true;
  render();
  try {
    cat = await api('/catalog' + (force ? '?refresh=1' : ''));
  } catch (e) { toast('خطا در دریافت کاتالوگ: ' + e.message, false); }
  catLoading = false;
  render();
}

async function openCountry(code) {
  if (!cat) return;
  catOpenCountry = code;
  if (!cat.byCountry[code]) {
    try {
      var r = await api('/catalog?country=' + encodeURIComponent(code));
      cat.byCountry[code] = r.proxies || [];
    } catch (e) { toast('خطا: ' + e.message, false); }
  }
  render();
}

async function addCatalogProxies(code) {
  var rows = (cat.byCountry[code] || []);
  if (!rows.length) return;
  var proxies = rows.map(function (p) { return p.proxy; });
  var existing = locs.find(function (l) { return l.code === code.toLowerCase(); });
  var name = (cat.countries.find(function (c) { return c.code === code; }) || {}).name || code;
  var merged = existing
    ? [existing.proxy].concat(existing.proxies || []).concat(proxies).filter(Boolean).filter(function (v, i, a) { return a.indexOf(v) === i; })
    : proxies;
  try {
    await api('/locations', 'POST', JSON.stringify({ code: code, name: name, proxies: merged }));
    toast(merged.length + ' پروکسی در لوکیشن ' + code.toUpperCase() + ' ذخیره شد ✓');
    await refresh();
  } catch (e) { toast('خطا: ' + e.message, false); }
}

async function saveSettingsForm() {

function render() {
  var v = $('view');
  var body;
  if (active === 'users') body = usersTab();
  else if (active === 'locs') body = locsTab();
  else if (active === 'subs') body = subsTab();    else if (active === 'catalog') body = catalogTab();
    else if (active === 'pref') body = prefTab();
    else if (active === 'latency') body = latencyTab();
  else body = settingsTab();
  v.innerHTML = (active !== 'settings' ? statCards() : '') + body;
  var tabs = document.querySelectorAll('.tabs button');
  for (var i = 0; i < tabs.length; i++) {
    tabs[i].classList.toggle('ghost', tabs[i].getAttribute('data-arg') !== active);
  }
}

async function refresh() {
  var d = await api('/state');
  users = d.users || [];
  locs = d.locations || [];
  subs = d.subs || [];
  settings = d.settings || settings;
  render();
  if (active === 'locs') loadLocationHealth();
}


// One delegated click handler resolves every data-act button (tabs, rows, dialogs).
// closest() so clicks on inner elements (icons, spans) still hit the handler.
document.addEventListener('click', function (e) {
  var el = e.target && e.target.closest ? e.target.closest('[data-act]') : null;
  if (!el) return;
  var act = el.getAttribute('data-act');
  if (!act) return;
  var arg = el.getAttribute('data-arg') || '';
  if (act === 'tab') { active = arg; render(); return; }
  if (act === 'user-new') return showUserDlg('');
  if (act === 'user-edit') return showUserDlg(arg);
  if (act === 'user-links') return showLinks(arg);
  if (act === 'user-toggle') return toggleUser(arg);
  if (act === 'user-del') return delUser(arg);
  if (act === 'loc-new') return showLocDlg('');
  if (act === 'loc-edit') return showLocDlg(arg);
  if (act === 'loc-del') return delLoc(arg);
  if (act === 'sub-new') return showSubDlg();
  if (act === 'sub-copy') return copySub(arg);
  if (act === 'sub-copy-sb') return copySub(arg, '?target=singbox');
  if (act === 'sub-copy-clash') return copySub(arg, '?target=clash');
  if (act === 'sub-del') return delSub(arg);
  if (act === 'save-user') return saveUser(editingUuid);
  if (act === 'save-loc') return saveLoc(editingLoc);
  if (act === 'save-sub') return saveSub();
  if (act === 'copy-all') return copyText(currentLinks.join('\n'));
  if (act === 'dlg-close') return closeDlg();
  if (act === 'cat-open') return openCountry(arg);
  if (act === 'cat-back') { catOpenCountry = ''; render(); return; }
  if (act === 'cat-refresh') return loadCatalog(true);
  if (act === 'cat-proto') { catProto = arg; openCountry(catOpenCountry); return; }
  if (act === 'cat-add') return addCatalogProxies(arg);
  if (act === 'pref-refresh') return loadPreferred(true);
  if (act === 'toggle-catalog') return toggleCatalogRouting();
  if (act === 'save-settings') return saveSettingsForm();
  if (act === 'run-latency') return runLatency();
  if (act === 'save-latency') return saveLatencyResults();
});

async function loadLocationHealth() {
  var badges = document.querySelectorAll('[data-health-code]');
  for (var i = 0; i < badges.length; i++) {
    var badge = badges[i];
    try {
      var r = await api('/location-health?code=' + encodeURIComponent(badge.getAttribute('data-health-code')));
      var live = (r.results || []).filter(function (x) { return x.ok; });
      badge.textContent = live.length ? ('پینگ: ' + live[0].ms + ' ms') : 'آفلاین';
      badge.className = 'pill ' + (live.length ? 'ok' : 'off');
    } catch (e) { badge.textContent = 'نامشخص'; }
  }
}


window.showUserDlg = function (uuid) {
  editingUuid = uuid || '';
  var u = editingUuid ? users.find(function (x) { return x.uuid === editingUuid; }) : null;
  // Locations = manual locations + live catalog countries (catalog wins for
  // display names). For a NEW user the top catalog countries are pre-checked
  // so every user is multi-location by default.
  var catMap = {};
  if (cat) for (var i = 0; i < cat.countries.length; i++) catMap[cat.countries[i].code.toLowerCase()] = cat.countries[i];
  var opts = [];
  var seenCodes = {};
  for (var j = 0; j < locs.length; j++) {
    var l = locs[j];
    seenCodes[l.code] = true;
    opts.push({ code: l.code, label: (l.code.toUpperCase() + ' — ' + (l.name || 'دستی')), sel: !!(u && u.countries && u.countries.indexOf(l.code) >= 0) });
  }
  var catCodes = Object.keys(catMap).sort(function (a, b) { return (catMap[b].total || 0) - (catMap[a].total || 0); }).slice(0, 40);
  for (var k = 0; k < catCodes.length; k++) {
    if (seenCodes[catCodes[k]]) continue;
    seenCodes[catCodes[k]] = true;
    var c = catMap[catCodes[k]];
    opts.push({ code: catCodes[k], label: (catCodes[k].toUpperCase() + ' — ' + (c.name || 'کاتالوگ') + ' (' + c.total + ')'), sel: u ? !!(u.countries && u.countries.indexOf(catCodes[k]) >= 0) : k < 5 });
  }
  var copts = opts.map(function (o) {
    return '<option value="' + escHtml(o.code) + '"' + (o.sel ? ' selected' : '') + '>' + escHtml(o.label) + '</option>';
  }).join('');
  openDlg('<h2>' + (u ? 'ویرایش کاربر' : 'کاربر جدید') + '</h2>' +
    '<div class="kv"><div><label>نام</label><input id="f-remark" value="' + (u ? escHtml(u.remark) : '') + '"></div>' +
    '<div><label>UUID (خالی = تصادفی)</label><input id="f-uuid" class="ltr" value="' + (u ? u.uuid : '') + '"' + (u ? ' readonly' : '') + '></div>' +
    '<div><label>حجم (GB — 0 = بی‌نهایت)</label><input id="f-limit" type="number" min="0" step="0.1" value="' + (u ? (u.limit_bytes / 1073741824) : 0) + '"></div>' +
    '<div><label>انقضا (روز — 0 = بی‌نهایت)</label><input id="f-expire" type="number" min="0" value="' + (u && u.expire ? Math.ceil((u.expire - Date.now() / 1000) / 86400) : 0) + '"></div>' +
    '<div><label>حد اتصال همزمان (0 = بی‌نهایت)</label><input id="f-cc" type="number" min="0" value="' + (u ? (u.concurrent_connections || 0) : 0) + '"></div>' +
    '<div><label>پروکسی اختصاصی (خالی = خودکار)</label><input id="f-proxyip" class="ltr" value="' + (u ? escHtml(u.proxy_ip || '') : '') + '"></div></div>' +
    '<label>لوکیشن‌ها (چندگزینه‌ای — دستی + کاتالوگ زنده؛ جدید: ۵ کشور برتر پیش‌فرض)</label>' +
    '<select id="f-countries" multiple size="8">' + copts + '</select>' +
    '<div class="actions" style="margin-top:14px"><button class="small" data-act="save-user">ذخیره</button>' +
    '<button class="small ghost" data-act="dlg-close">انصراف</button></div>');
};
